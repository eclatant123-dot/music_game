        let user = null;

            try {

                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {

                    await signInWithCustomToken(auth, __initial_auth_token);

                    await signInAnonymously(auth);

            if(!user) return;

            try {

                const dbRef = collection(db, 'artifacts', appId, 'public', 'data', 'leaderboard');

                await addDoc(dbRef, {

                    nickname: nickname,

                    score: score,

                    timestamp: Date.now()

                });

            } catch(e) {

                console.error("Save error", e);

            }

        };


        let unsubscribe = null;

        window.listenToLeaderboard = (callback) => {

            if(!user) {

                callback([]);

                return;

            }

            const dbRef = collection(db, 'artifacts', appId, 'public', 'data', 'leaderboard');

            if (unsubscribe) unsubscribe();

            unsubscribe = onSnapshot(dbRef, (snapshot) => {

                let docs = [];

                snapshot.forEach(doc => docs.push(doc.data()));

                docs.sort((a,b) => b.score - a.score);

                callback(docs.slice(0, 50)); 

            }, (error) => {

                console.error("Listen error", error);

                callback([]);

            });

        };

        window.stopLeaderboardListen = () => {

            if (unsubscribe) unsubscribe();

        };

        function loadNextNote() {

            if (currentQuestion >= MAX_QUESTIONS) {

                endGame();

                return;

            }

            currentQuestion++;

            document.getElementById('questionCount').innerText = `${currentQuestion} / ${MAX_QUESTIONS}`;


            const sourceArray = currentMode === 'treble' ? notesTreble : notesBass;

            let nextNote = sourceArray[Math.floor(Math.random() * sourceArray.length)];

            

            // Prevent same note twice in a row

            while(currentNote && nextNote.pitch === currentNote.pitch) {

                nextNote = sourceArray[Math.floor(Math.random() * sourceArray.length)];

            }

            currentNote = nextNote;


            drawNote(currentNote);

            startTimer();

        }

        function startTimer() {

            stopTimer();

            startTime = Date.now();

            timerInterval = requestAnimationFrame(updateTimer);

        }


        function updateTimer() {

            const elapsed = Date.now() - startTime;

            const remaining = Math.max(0, TIME_LIMIT - elapsed);

            const percentage = (remaining / TIME_LIMIT) * 100;

            

            const timerBar = document.getElementById('timerBar');

            timerBar.style.width = percentage + '%';

            

            if (percentage > 50) {

                timerBar.className = "h-full w-full rounded-full bg-gradient-to-r from-indigo-400 to-purple-500 transition-all duration-75";

            } else if (percentage > 20) {

                timerBar.className = "h-full w-full rounded-full bg-amber-400 transition-all duration-75";

            } else {

                timerBar.className = "h-full w-full rounded-full bg-rose-500 transition-all duration-75";

            }

            

            if (remaining > 0) {

                timerInterval = requestAnimationFrame(updateTimer);

            } else {

                handleTimeOut();

            }

        }


        function stopTimer() {

            if (timerInterval) {

                cancelAnimationFrame(timerInterval);

            }

            document.getElementById('timerBar').style.width = '0%';

        }


        function handleTimeOut() {

            if (isProcessing) return;

            isProcessing = true;

            

            combo = 0;

            playWrongSound();

            updateScoreUI();

            

            const staffContainer = document.getElementById('staffContainer');

            staffContainer.classList.add('shake-animation');

            

            setTimeout(() => {

                staffContainer.classList.remove('shake-animation');

                isProcessing = false;

                loadNextNote();

            }, 400);

        }


        function skipNote() {

            if (isProcessing) return;

            combo = 0;

            updateScoreUI();

            stopTimer();

            loadNextNote();

        }


        function checkAnswer(answerName) {

            if (isProcessing || !currentNote) return;

            isProcessing = true;

            stopTimer();


            const isCorrect = answerName === currentNote.name;

            const staffContainer = document.getElementById('staffContainer');


            if (isCorrect) {

                const elapsed = Date.now() - startTime;

                const remaining = Math.max(0, TIME_LIMIT - elapsed);

                const timeBonus = Math.floor((remaining / TIME_LIMIT) * 10);


                combo++;

                const points = 10 + timeBonus + (Math.floor(combo / 2) * 5);

                score += points;

                

                // Play Piano Sound of the Correct Note (A4 = 440Hz)

                playPianoCorrectSound(currentNote.pitch, combo);

                showScoreFeedback(points);

                spawnFireworks();

                updateScoreUI();


                setTimeout(() => {

                    isProcessing = false;

                    loadNextNote();

                }, 450);


            } else {

                combo = 0;

                playWrongSound();

                updateScoreUI();

                

                staffContainer.classList.add('shake-animation');

                setTimeout(() => {

                    staffContainer.classList.remove('shake-animation');

                    isProcessing = false;

                    loadNextNote();

                }, 400);

            }

        }

        function playPianoCorrectSound(pitch, currentCombo) {

            if(!pianoSynth) return;

            const now = Tone.now();

            

            pianoSynth.triggerAttackRelease(pitch, "2n", now);

            

            if (currentCombo > 0 && currentCombo % 3 === 0) {

                cheerFilter.frequency.value = 300;

                cheerNoise.triggerAttackRelease("1n", now + 0.05);

                cheerFilter.frequency.rampTo(2500, 0.4, now + 0.05);

            }

        }

        class Particle {

            constructor(x, y, color) {

                this.x = x;

                this.y = y;

                this.color = color;

                const angle = Math.random() * Math.PI * 2;

                const speed = Math.random() * 5 + 2;

                this.vx = Math.cos(angle) * speed;

                this.vy = Math.sin(angle) * speed;

                this.alpha = 1;

                this.decay = Math.random() * 0.02 + 0.015;

                this.size = Math.random() * 4 + 2;

            }

            update() {

                this.x += this.vx;

                this.y += this.vy;

                this.vy += 0.1;

                this.alpha -= this.decay;

            }

            draw(ctx) {

                ctx.save();

                ctx.globalAlpha = this.alpha;

                ctx.fillStyle = this.color;

                ctx.beginPath();

                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);

                ctx.fill();

                ctx.restore();

            }

        }


        function spawnFireworks() {

            const colors = ['#f43f5e', '#fbbf24', '#34d399', '#60a5fa', '#a78bfa', '#f472b6'];

            const staffRect = document.getElementById('staffContainer').getBoundingClientRect();

            const appRect = document.querySelector('.app-container').getBoundingClientRect();

            

            const x = (staffRect.left - appRect.left) + staffRect.width / 2;

            const y = (staffRect.top - appRect.top) + staffRect.height / 2;


            for(let i=0; i<30; i++) {

                const color = colors[Math.floor(Math.random() * colors.length)];

                particles.push(new Particle(x, y, color));

            }

        }


        function animateFireworks() {

            ctx.clearRect(0, 0, canvasWidth, canvasHeight);

            

            for(let i = particles.length - 1; i >= 0; i--) {

                const p = particles[i];

                p.update();

                p.draw(ctx);

                if(p.alpha <= 0) {

                    particles.splice(i, 1);

                }

            }

            

            requestAnimationFrame(animateFireworks);

        }

            resizeCanvas();

            animateFireworks();

        }
